const UI_TEST_RESULTS = { passed: [], failed: [], skipped: [] }; function logTest(name, passed, error = null) { if (passed) { UI_TEST_RESULTS.passed.push(name); console.log(`✅ PASS: ${name}`); } else { UI_TEST_RESULTS.failed.push({ name, error }); console.error(`❌ FAIL: ${name}`, error); } } function logSkip(name, reason) { UI_TEST_RESULTS.skipped.push({ name, reason }); console.warn(`⏭️ SKIP: ${name} - ${reason}`); } async function sleep(ms) { return new Promise(resolve => setTimeout(resolve, ms)); } async function runAllTests() { console.log("🚀 Starting Comprehensive UI Tests...\n"); try { // Test App Header console.log("\n=== Testing App Header ==="); const toggleBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Side B') || b.textContent.includes('Side A') ); if (toggleBtn) { toggleBtn.click(); await sleep(500); logTest("Toggle Side A/B", true); toggleBtn.click(); await sleep(500); logTest("Toggle back to Side A", true); } // Test Side B console.log("\n=== Testing Side B ==="); if (toggleBtn && toggleBtn.textContent.includes('Side A')) { toggleBtn.click(); await sleep(500); } // Test Load Emotions const loadBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Load Emotions') ); if (loadBtn && !loadBtn.disabled) { loadBtn.click(); await sleep(2000); logTest("Load Emotions", true); } else { logSkip("Load Emotions", loadBtn ? "Button disabled" : "Button not found"); } // Test Emotion Wheel await sleep(1000); const baseBtns = document.querySelectorAll('.emotion-wheel-grid-base button'); if (baseBtns.length > 0) { console.log("\n--- Testing Emotion Wheel ---"); baseBtns[0].click(); await sleep(500); logTest("Select base emotion", true); const intensityBtns = document.querySelectorAll('.emotion-wheel-grid-intensity button'); if (intensityBtns.length > 0) { intensityBtns[0].click(); await sleep(500); logTest("Select intensity level", true); const subBtns = document.querySelectorAll('.emotion-wheel-grid-sub button'); if (subBtns.length > 0) { subBtns[0].click(); await sleep(500); logTest("Select sub-emotion", true); // Test Clear button const clearBtn = document.querySelector('.emotion-wheel-clear-btn'); if (clearBtn) { clearBtn.click(); await sleep(300); logTest("Clear emotion selection", true); } } } } else { logSkip("Emotion Wheel", "No emotions loaded"); } // Test Generate Music const genBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Generate Music') && !b.disabled ); if (genBtn) { genBtn.click(); await sleep(2000); logTest("Generate Music", true); } else { logSkip("Generate Music", "Button not available or requires emotion"); } // Test Interrogation const interrogateBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Interrogation') ); if (interrogateBtn && !interrogateBtn.disabled) { interrogateBtn.click(); await sleep(2000); logTest("Start Interrogation", true); } else { logSkip("Start Interrogation", interrogateBtn ? textarea.dispatchEvent(new Event('input', { bubbles: true })); await sleep(300); } const saveBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Save Lyrics') ); if (saveBtn) { saveBtn.click(); await sleep(1000); logTest("Save lyrics", true); } const refreshBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Refresh') ); if (refreshBtn) { refreshBtn.click(); await sleep(1000); logTest("Refresh lyrics", true); } const clearBtn = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Clear') && b.closest('.lyric-panel') ); if (clearBtn) { clearBtn.click(); await sleep(1000); logTest("Clear lyrics", true); } // Test Guide Navigation console.log("\n=== Testing Guide Navigation ==="); const searchInput = document.querySelector('.guide-search'); if (searchInput) { searchInput.value = "test"; await sleep(300); } const topicPills = document.querySelectorAll('.topic-pill'); if (topicPills.length > 0) { const allBtn = Array.from(topicPills).find(b => b.textContent.trim() === 'All'); if (allBtn) { allBtn.click(); if (previewBtn) { previewBtn.click(); await sleep(500); logTest("Preview guide", true); } const copyBtn = document.querySelector('.copy-btn'); if (copyBtn) { copyBtn.click(); await sleep(300); logTest("Copy guide path", true); } // Test SpectoCloud Panel console.log("\n=== Testing SpectoCloud Panel ==="); const loadConfig = Array.from(document.querySelectorAll('button')).find(b => b.textContent.includes('Load Humanizer Config') ); Type 'runUITests()' to start testing.");
